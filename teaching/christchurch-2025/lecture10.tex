\begin{frame}{Publication bias $>$ attenuation bias}
\begin{itemize}
\item $|$IV$|$ $>$ $|$OLS$|$ $\rightarrow$ attenuation or other endogeneity biases important
	\item OLS = Natural experiments $\rightarrow$ other endogeneity biases negligible
	\item[$\rightarrow$] \alert{$|$IV $-$ OLS$|$ measures attenuation bias, which is strong}
	\item Uncorrected inverted elasticity = $-0.67$
	\item Inverted elasticity corrected for pub and attenuation bias (IV) = $-0.25$
	\item[$\rightarrow$] Pub bias dominates attenuation
	\item[$\rightarrow$] Implied elasticity = 4 
\end{itemize}
\end{frame}

\begin{frame}{Example 2: Too much heterogeneity?}
    
    \textbf{Common measure:} estimate the relative heterogeneity of underlying effects \textit{(Higgins and Thompson, 2002)}
    
    \[
    I^2 \equiv \frac{\text{underlying variance of effects}}{\text{observed variance of estimates}}
    \]
    
    \begin{itemize}
        \item $I^2 = 0.75 \sim 1.0$ \dots \textit{``considerable heterogeneity'' in medicine (Cochrane Review Guideline)}
        \item $I^2 \simeq 0.9$ \dots common in economics meta-analyses (e.g., \textit{Vivalt, 2021})
    \end{itemize}
    
    \bigskip
    
    \textbf{Unresolved concern:}
    \begin{itemize}
        \item no universally agreed criteria for ``too much'' heterogeneity
        \item undesirable pattern: with more precise studies, higher $I^2$
    \end{itemize}
\end{frame}


\begin{frame}{This Paper}
    \begin{enumerate}
        \item Propose a new test for external validity in meta-analysis:
        \begin{itemize}
            \item Based on the tail index of the underlying distribution, without contamination by estimate precision.
            \item Test whether the implied variance of underlying effect is infinite (i.e., too heterogeneous).
        \end{itemize}
        
        \bigskip
        
        \item Show that the implied variance is infinite in meta-analysis of 126 nudge RCTs \textit{(DellaVigna and Linos, 2020)}:
        \begin{itemize}
            \item \textbf{Nudge:} ``choice architecture that alters people's behavior in a predictable way without forbidding any options or significantly changing their economic incentives.'' \textit{(Thaler and Sunstein, 2008)}
            \item The possibility of infinite variance can be important in real-world applications.
        \end{itemize}
    \end{enumerate}
\end{frame}

\begin{frame}{Test of Infinite Variance}
    \textbf{Pareto tail:}
    \[
    F(b) = 1 - Cb^{-\alpha}, \quad b > b_{\text{min}} \quad \text{and} \quad C > 0
    \]

    \begin{itemize}
        \item e.g. earthquakes, word frequency, city size, wealth distribution
        \item If $\alpha < 2$, then $V\!ar(b) = \infty$ because
        \[
        \int_{b_{\text{min}}}^\infty b^2 f(b) \, db \propto \int_{b_{\text{min}}}^\infty b^2 b^{-3} \, db = \int_{b_{\text{min}}}^\infty b^{-1} \, db = \log b \big|_{b_{\text{min}}}^\infty = \infty
        \]
    \end{itemize}
    
    \bigskip
    
    \textbf{Tail Index Test of External Validity:}
    \begin{itemize}
        \item \textbf{Criteria 1:} if $\hat{\alpha} > 2$, then pass the test
        \item \textbf{Criteria 2:} if $\mathbb{P}(\hat{\alpha} \leq 2) < p$, then pass the test
    \end{itemize}
\end{frame}


\begin{frame}{Estimation Methods (i) Inner Problem of Tail Index}
    Following \textit{Hill (1975)}, use Maximum Likelihood Estimation for $\hat{\alpha}_{MLE}$: given $b_{\text{min}}$,
    \[
    \hat{\alpha}_{MLE} = \arg \max_{\alpha} \sum_{i=1}^n \ln l \left( \alpha \mid \beta_i, SE_i, b_{\text{min}} \right),
    \]
    where
    \[
    l \left( \alpha \mid \beta_i, SE_i, b_{\text{min}} \right) \equiv \int_{\theta_{\text{min}}}^\infty
    \frac{\phi \left( \frac{\beta_i - \theta}{SE_i} \right)}{1 - \Phi \left( \frac{b_{\text{min}} - \theta}{SE_i} \right)}
    \alpha b_{\text{min}}^{\alpha} b^{-(\alpha + 1)} \, db.
    \]

    \bigskip

    \begin{itemize}
        \item Numerical integration
    \end{itemize}
\end{frame}


\begin{frame}{Estimation Methods (ii) Outer Problem of Cut-off}
    Following \textit{Clauset et al. (2006)}, use the goodness-of-fit metric to choose $b^*_{\text{min}}$:
    \[
    b^*_{\text{min}} \in \arg \min_{b_{\text{min}}} \tilde{K}S \left(b_{\text{min}} \mid \hat{\alpha}_{MLE}\right),
    \]
    where the reweighted Kolmogorov-Smirnov statistic is
    \[
    \tilde{K}S \left(b_{\text{min}} \mid \hat{\alpha}_{MLE}\right) \equiv \sup_{\beta} 
    \frac{\left|G(\beta \mid \beta \geq b_{\text{min}}) - \hat{G}(\beta \mid \cdot)\right|}
    {\sqrt{\hat{G}(\beta \mid \cdot) \left[1 - \hat{G}(\beta \mid \cdot)\right]}},
    \]
    and the estimated distribution is $\hat{G}(\beta \mid \hat{\alpha}_{MLE}, \beta \geq b_{\text{min}})$ and the empirical distribution is $G(\beta \mid \beta \geq b_{\text{min}})$.
\end{frame}

\begin{frame}{Summary}
    \begin{enumerate}
        \item Proposed the concept of \textit{``infinite variance''} as representing the persistent criticism that \textit{``too much'' heterogeneity} underlies the overall mean estimate.
        
        \bigskip
        
        \item Developed an estimation method by extending the maximum likelihood estimation and goodness-of-fit statistics commonly used in tail estimation.
        
        \bigskip
        
        \item Applied the estimation method and its implied test to show that the influential nudge meta-analysis exhibits the infinite variance.
    \end{enumerate}
\end{frame}
