\begin{frame}{Delta method}
Based on Taylor series expansion. Basic formula:

\[
\text{Var}(Y) = \text{Var}(f(X)) \approx [f'(\mu_X)]^2 \text{Var}(X).
\]

% So, if $Y$ is any function of a random variable $X$, we need only calculate the variance of $X$ and the first derivative of the function to approximate the variance of $Y$.

\bigskip

\underline{\text{Example:}} \, \text{Suppose } $Y = X^2$. \, \text{Then } $f(x) = x^2$ \, \text{and } $f'(x) = 2x$, \, \text{so that:}
\[
\text{Var}(Y) \approx [2\mu_X]^2 \text{Var}(X) = 4\mu_X^2\sigma_X^2.
\]

\underline{\text{Example:}} \, \text{Suppose } $Y = \frac{1}{X}$. \, \text{Then } $f(x) = \frac{1}{x}$ \, \text{and } $f'(x) = -\frac{1}{x^2}$, \, \text{so that:}
\[
\text{Var}(Y) \approx \left[\frac{-1}{\mu_X^2}\right]^2 \text{Var}(X) = \frac{\sigma_X^2}{\mu_X^4}.
\]



\end{frame}
