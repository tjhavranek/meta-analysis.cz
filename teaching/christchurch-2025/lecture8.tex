\begin{frame}{Key difference}
\begin{itemize}
	\item \textbf{Publication bias}: all estimates are individually unbiased
	\item The entire literature is biased because estimates are selectively reported
	\item If we know publication probabilities, we can recover the true effect (selection models)
	\item \textbf{P-hacking:} individual estimates can be biased
	\item Classical selection models fail
	\item Funnel-based models can work (with adjustments)
\end{itemize}
\end{frame}

\begin{frame}{Options for the meta-analyst}
\begin{enumerate}

	\item Remove ``bad'' variation from SE $\rightarrow$ MAIVE.

\begin{frame}{Bonus: Maya Mathur's p-hacking correction}
\begin{itemize}
	\item Right-trunctated meta-analysis (RTMA, Mathur 2024, RSM)
	\item Assumption 1: insignificant estimates are NOT p-hacked
	\item Assumption 2: estimates are normally distributed
	\item Assumption 3: there are enough insignificant estimates published that we can recover the underlying distribution
	\item Bayesian techniques used for better convergence
