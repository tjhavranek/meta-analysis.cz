# The data layer is only reproducible if these are pinned.
#
# pandas and pyarrow are embedded in every Parquet file this pipeline writes: pyarrow in
# `created_by`, pandas in the file's pandas metadata blob. A different build therefore
# changes all 45 per-dataset checksums with no change to a single value — which would tell
# anyone who pinned a hash that the data moved when it had not.
#
# Measured 2026-08-21 in a clean clone: unpinned, a rebuild differed from the release in
# 47 of 160 files (46 Parquet plus croissant.json), every one of them version metadata
# rather than data. With these pins the rebuild is byte-identical, and it is byte-identical
# on Python 3.11 and 3.12 alike — the interpreter version does not matter, these pins do.
pandas==3.0.3
pyarrow==25.0.0

# The Excel readers. Not optional: 25 of the 45 source archives are .xls/.xlsx/.xlsm, and
# without these `06_convert.py` fails on 12 datasets. They were omitted until 2026-08-21,
# so anyone who followed this file literally could not reproduce the release at all —
# which is the one thing the file exists to guarantee. xlrd reads .xls, openpyxl the rest.
xlrd==2.0.2
openpyxl==3.1.5

# NumPy is imported directly by 06_convert, 08_harmonise, 09_verify and the 9x gates, and it
# does the numerical work behind every published statistic. pandas constrains it loosely, which
# is not the same as pinning it: a future install could resolve a different build and move a
# value without moving a version number anyone reads.
numpy==2.4.6
